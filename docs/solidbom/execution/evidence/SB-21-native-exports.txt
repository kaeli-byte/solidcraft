def truncate(text, limit=6000):

def raw_request(method, path, body=None, token=TOKEN, raw_bytes=None):
    headers = {"Authorization": f"Bearer {token}"}
    if raw_bytes is not None:
        data = raw_bytes
    elif body is not None:
        data = json.dumps(body).encode()
        headers["Content-Type"] = "application/json"
    else:
        data = None
    req = urllib.request.Request(BASE + path, data=data, method=method, headers=headers)
    try:
        with urllib.request.urlopen(req) as response:
            return response.status, response.read().decode()
    except urllib.error.HTTPError as error:
        return error.code, error.read().decode()


def request(method, path, body=None, token=TOKEN, capture=None, title=None, raw_bytes=None, expect=200):

        raise SystemExit(1)
    return json.loads(text) if text else None

def start_service():
    log = open(os.path.join(OUT, "service.log"), "wb")
    process = subprocess.Popen([BIN], cwd=os.path.join(REPO, "services", "solidbom-server"), env=service_env(), stdout=log, stderr=subprocess.STDOUT)
    deadline = time.time() + 60
    while time.time() < deadline:
        if process.poll() is not None:
            log.close()
            with open(os.path.join(OUT, "service.log"), "rb") as handle:
                raise SystemExit(f"service exited during startup:\n{handle.read().decode(errors='replace')[-4000:]}")
        try:
            with urllib.request.urlopen(BASE + "/health", timeout=2) as response:
                if response.status == 200 and json.loads(response.read().decode()) == {"ok": True}:
                    return process, log
        except Exception:
            time.sleep(0.25)
    process.terminate()
    raise SystemExit("service did not become healthy within 60 s")


def stop_service(process, log):
    if process.poll() is None:
        process.send_signal(signal.SIGTERM)
        try:
            process.wait(timeout=15)
        except subprocess.TimeoutExpired:
            process.kill()
            process.wait(timeout=10)
    log.close()

def main():
    if os.path.isdir(OUT):
        shutil.rmtree(OUT)
    os.makedirs(OUT, exist_ok=True)
    started = time.time()
    with open(BIN, "rb") as handle:
        binary_sha256 = hashlib.sha256(handle.read()).hexdigest()
    binary_mtime = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(os.path.getmtime(BIN)))
    recreate_database()
    process, log = start_service()
    try:
        check("service health is ok", True)
        check("the service reports the engineer identity", ok("GET", "/v1/me")["name"] == "development",
              ok("GET", "/v1/me")["name"])
        source = None
        complete = seed_complete_project()
        # The source ref is only needed again by the second project; re-upload is idempotent.
        status, text = raw_request("PUT", f"/v1/sources/{SHA}?name=sb21-evidence.step", raw_bytes=SOURCE)
        check("the source re-upload is idempotent", status == 200, text[:500])
        source = json.loads(text)
        incomplete = seed_incomplete_project(source)
        summary = {
            "database": DB,
            "database_url_user": USER,
            "base_url": BASE,
            "service_binary": BIN,
            "service_binary_sha256": binary_sha256,
            "service_binary_mtime": binary_mtime,
            "source_sha256": SHA,
            "project_complete": complete["project"],
            "project_incomplete": incomplete["project"],
            "complete": {
                "baseline_one": complete["baseline_one"],
                "accepted_sequence": complete["accepted_one"]["sequence"],
                "accepted_calculation_id": complete["accepted_one"]["costing_calculation_id"],
                "parts": [entry["part_version"] for entry in complete["scoped"]],
                "products": [entry["product"]["product_id"] for entry in complete["scoped"]],
                "component_id": complete["component_id"],
                "component_v1": complete["component_v1"],
                "material_id": complete["material"]["material_id"],
                "material_v1": complete["material"]["material_version_id"],
                "production_cost_center_id": complete["production"]["cost_center_id"],
                "production_rate_v1": complete["production"]["rate_version_id"],
            },
            "incomplete": {
                "calculation_id": incomplete["calculation_id"],
                "labels": [],
            },
        }
        if SEEDING_ONLY:
            summary["mode"] = "seeding-only"
            save_json("seeding-summary.json", summary)
            print(json.dumps({"checks": len(CHECKS), "last_checks": CHECKS[-6:], "summary": summary}, indent=2))
            return

        run_export_observations(complete, incomplete, summary)
        save_json("captures.json", {"captures": CAPTURES, "checks": CHECKS, "documents": DOCS, "summary": summary,
                                    "defects": DEFECTS, "notes": NOTES,
                                    "duration_seconds": round(time.time() - started, 1)})
        print(json.dumps({"checks": len(CHECKS), "documents": {name: {"file_name": doc["file_name"], "byte_len": doc["byte_len"],
                                                                     "sha256": doc["sha256"], "labels": doc["metadata"]["labels"]}
                                                              for name, doc in DOCS.items()}}, indent=2))
    finally:
        stop_service(process, log)
        # The database is disposable and this script owns it: stop the service first, then drop it.
        try:
            run(["dropdb", "--if-exists", DB], env=pg_env())
        except subprocess.CalledProcessError as error:
            print(f"warning: could not drop the disposable database {DB}: {error}", file=sys.stderr)

def dec(value):
    if value is None:
        return None
    if isinstance(value, Decimal):
        return value
    if isinstance(value, bool):
        return None
    if isinstance(value, (int, float)):
        return Decimal(str(value))
    return Decimal(str(value))


def close(actual, expected, tolerance=None):
    actual = dec(actual)
    expected = dec(expected)
    if actual is None or expected is None:
        return actual is expected
    if tolerance is None:
        tolerance = max(Decimal("0.000001"), abs(expected) * Decimal("1e-10"))
    return abs(actual - expected) <= tolerance


def money_text(value):
    """The PDF renderer's money format: two fractional digits, grouped in threes."""
    rounded = dec(value).quantize(Decimal("0.01"))
    negative = rounded < 0
    scaled = int(abs(rounded) * 100)
    whole, fraction = divmod(scaled, 100)
    grouped = f"{whole:,}"
    return f"{'-' if negative else ''}{grouped}.{fraction:02d}"


def shown_money(value):
    """The Decimal a printed money string carries (grouping separators removed)."""
    return Decimal(money_text(value).replace(",", ""))


def number_tokens(text):
    tokens = []
    for match in re.findall(r"-?\d[\d,]*\.\d{2}(?!\d)", text):
        tokens.append(Decimal(match.replace(",", "")))
    return tokens


def page_texts(reader):

def rows_of(worksheet):
    return [[cell.value for cell in row] for row in worksheet.iter_rows()]


def section(rows, title, occurrences=1):
    """The rows after a Costs/Tooling section title, up to the next blank row."""
    seen = 0
    for index, row in enumerate(rows):
        if row and row[0] == title:
            seen += 1
            if seen == occurrences:
                data = []
                for follow in rows[index + 1 :]:
                    if all(value is None for value in follow):
                        break
                    data.append(follow)
                return data
    raise AssertionError(f"section {title!r} not found")


def has_number_format(worksheet, expected_text):
    for row in worksheet.iter_rows():
        for cell in row:
            if isinstance(cell.value, (int, float)) and not isinstance(cell.value, bool) and cell.number_format == expected_text:
                return cell.coordinate
    return None


def workbook_values(workbook):
    """Every numeric cell value of the workbook, as Decimal strings, for pair comparison."""
    values = []

            for value in row:
                if isinstance(value, (int, float)) and not isinstance(value, bool):

    return values

        check(f"{kind}: the Products row keeps the stored schedule", row[7] == stored["schedule"]["sop_year"] and row[8] == stored["schedule"]["eop_year"], row[7:9])
        for position, year in enumerate(expected_years):
            forecast = next((entry["quantity"] for entry in stored["forecasts"] if entry["year"] == year), None)
            check(f"{kind}: the Products {year} forecast column keeps the stored forecast",
                  row[15 + position] == forecast, (product_id, year, row[15 + position], forecast))
    if kind != "incomplete":
        check(f"{kind}: the workbook carries the exact non-Latin product name",
              any("支架总成 Support bracket assembly (非 Latin 名称)" in (cell or "") for row in product_rows for cell in row if isinstance(cell, str)),
              "missing unicode product name")

    # -- BOM: one row per product x part version --------------------------------------------
    bom = workbook["BOM"]
    bom_rows = rows_of(bom)
    bom_headers = bom_rows[0]
    check(f"{kind}: the BOM headers are the contract columns", bom_headers[:10] ==
          ["Product ID", "Product Name", "Part Version ID", "Customer Part Number", "Customer Revision", "Classification",
           "Classification Confirmed", "Sourcing", "Quantity in Product", "Lifetime Demand"], bom_headers[:10])
    bom_years = [header for header in bom_headers if str(header).startswith("Demand ")]
    check(f"{kind}: the BOM demand columns follow the stored demand years",
          bom_years == [f"Demand {year}" for year in sorted({demand["year"] for part in outputs["parts"] for demand in part["annual_demand"]})],
          bom_years)
    check(f"{kind}: the BOM ends with the unit cost column", bom_headers[-1] == "Unit Cost", bom_headers[-1])
    bom_data = [row for row in bom_rows[1:] if row[0] is not None]
    expected_bom = sum(len(product["parts"]) for product in outputs["products"])
    check(f"{kind}: the BOM row count follows the product structures", len(bom_data) == expected_bom, (len(bom_data), expected_bom))
    for row in bom_data:
        product = products_out[row[0]]
        line = next((entry for entry in product["parts"] if entry["part_version_id"] == row[2]), None)
        check(f"{kind}: every BOM row is a product part line", line is not None, (row[0], row[2]))
        check(f"{kind}: the BOM quantity in one product is the stored quantity", row[8] == line["quantity"], (row[8], line["quantity"]))
        part = parts_out[row[2]]
        check(f"{kind}: the BOM unit cost is the stored part unit cost", close(row[-1], part["unit_cost"]), (row[-1], part["unit_cost"]))
        for position, header in enumerate(bom_years):
            year = int(str(header).split()[-1])
            demand = next((entry["quantity"] for entry in part["annual_demand"] if entry["year"] == year), None)
            check(f"{kind}: the BOM {year} demand keeps the stored per-part demand", row[10 + position] == demand,
                  (row[2], year, row[10 + position], demand))

    # -- Routings: part-root and product-root operations ------------------------------------
    routings = workbook["Routings"]
    routing_rows = rows_of(routings)
    routing_headers = routing_rows[0]
    check(f"{kind}: the Routings headers are the contract columns", len(routing_headers) == 15 and routing_headers[0] == "Owner Kind", routing_headers[:4])
    expected_operations = []
    for part in inputs["parts"]:
        for operation in part["operations"]:
            expected_operations.append(("part version", part["part_version_id"], operation))
    for product in inputs["products"]:
        for operation in product["root_operations"]:
            expected_operations.append(("product", product["product_id"], operation))
    routing_data = [row for row in routing_rows[1:] if row[0] is not None]
    check(f"{kind}: the Routings row count follows the stored operations", len(routing_data) == len(expected_operations),
          (len(routing_data), len(expected_operations)))
    seen_operations = set()
    for row in routing_data:
        key = (row[0], row[1], row[3])
        expected = next(((kind_name, owner_id, operation) for kind_name, owner_id, operation in expected_operations
                         if kind_name == row[0] and owner_id == row[1] and operation["sequence"] == row[3]), None)
        check(f"{kind}: every Routings row is a stored operation", expected is not None, key)
        operation = expected[2]
        seconds = operation["cycle"]["seconds"] if operation["cycle"] else None
        hours = operation["setup"]["hours"] if operation["setup"] else None
        check(f"{kind}: the routing cycle seconds keep the stored cycle", close(row[5], seconds), (key, row[5], seconds))
        check(f"{kind}: the routing setup hours keep the stored setup", close(row[11], hours), (key, row[11], hours))
        seen_operations.add(key)
    check(f"{kind}: every stored operation appears once", len(seen_operations) == len(expected_operations), len(seen_operations))

    # -- Feasibility ------------------------------------------------------------------------
    feasibility = workbook["Feasibility"]
    feasibility_rows = rows_of(feasibility)
    feasibility_data = [row for row in feasibility_rows[1:] if row[0] is not None]
    component_ids = {row[0] for row in feasibility_data}
    if kind == "incomplete":
        check(f"{kind}: the unfinished project has no feasibility finding", len(feasibility_data) == 0, len(feasibility_data))
    else:
        check(f"{kind}: the Feasibility sheet covers the reviewed components", len(component_ids) == 4, component_ids)
        check(f"{kind}: every finding in the complete project is a pass",
              all(row[4] == "pass" for row in feasibility_data),
              [(row[0], row[3], row[4]) for row in feasibility_data if row[4] != "pass"][:8])
        check(f"{kind}: the passing assessment records every rule finding", len(feasibility_data) >= 16, len(feasibility_data))

    # -- Costs: blocks, setup lines, annual series and totals -------------------------------
    costs = workbook["Costs"]
    cost_rows = rows_of(costs)
    part_lines = section(cost_rows, "Part cost lines")
    check(f"{kind}: the part cost lines follow the stored parts", len(part_lines) - 1 == len(outputs["parts"]), len(part_lines) - 1)
    for row in part_lines[1:]:
        part = parts_out.get(row[1])
        check(f"{kind}: every part cost line names a stored part", part is not None, row[1])
        check(f"{kind}: the part cost line unit cost is the stored unit cost", close(row[9], part["unit_cost"]), (row[1], row[9], part["unit_cost"]))
        check(f"{kind}: the part cost line setup charge is the stored setup charge", close(row[10], part["setup_charge"]), (row[1], row[10], part["setup_charge"]))
    setup_lines = section(cost_rows, "Separate setup lines")
    setup_data = setup_lines[1:]
    check(f"{kind}: the separate setup lines follow the stored setup lines", len(setup_data) == len(outputs["setup"]), len(setup_data))
    stored_setup = {(line["owner"].get("part_version_id") or line["owner"].get("product_id")): dec(line["setup_charge"])
                    for line in outputs["setup"]}
    workbook_setup_total = Decimal(0)
    for row in setup_data:
        owner_id = row[1]
        expected_charge = stored_setup.get(owner_id)
        check(f"{kind}: every setup line keeps the stored owner and charge", expected_charge is not None and close(row[2], expected_charge),
              (owner_id, row[2], expected_charge))
        workbook_setup_total += dec(row[2] or 0)
    annual_lines = section(cost_rows, "Annual project manufacturing cost")
    annual_data = annual_lines[1:]
    check(f"{kind}: the annual cost rows follow the stored calendar years", len(annual_data) == len(outputs["annual"]), len(annual_data))
    workbook_annual_sum = Decimal(0)
    for row, stored in zip(annual_data, outputs["annual"]):
        check(f"{kind}: the annual row year is the stored year {stored['year']}", row[0] == stored["year"], (row[0], stored["year"]))
        check(f"{kind}: the annual manufacturing cost for {stored['year']} reconciles to the snapshot",
              close(row[1], stored["manufacturing_cost"]), (stored["year"], row[1], stored["manufacturing_cost"]))
        if stored["manufacturing_cost"] is not None:
            workbook_annual_sum += dec(row[1])
    totals = section(cost_rows, "Project totals")
    totals_map = {row[0]: row[3] for row in totals if row and row[0] is not None}
    lifetime = totals_map.get("Lifetime manufacturing cost including separate setup")
    setup_total = totals_map.get("Separate setup total")
    check(f"{kind}: the workbook lifetime manufacturing total reconciles to the stored snapshot",
          close(lifetime, outputs["project_total_manufacturing_cost"]), (lifetime, outputs["project_total_manufacturing_cost"]))
    check(f"{kind}: the workbook separate setup total reconciles to the stored snapshot",
          close(setup_total, outputs["setup_total"]), (setup_total, outputs["setup_total"]))
    check(f"{kind}: the workbook setup lines sum to the workbook setup total", close(workbook_setup_total, setup_total),
          (workbook_setup_total, setup_total))
    if outputs["project_total_manufacturing_cost"] is not None:
        check(f"{kind}: the workbook annual series plus setup equals its lifetime total",
              close(workbook_annual_sum + dec(setup_total), lifetime), (workbook_annual_sum, setup_total, lifetime))
    check(f"{kind}: the setup-scope category block is empty when no category is stored",
          len(section(cost_rows, "Setup categories")) == 1, len(section(cost_rows, "Setup categories")))
    check(f"{kind}: the part purchase prices follow the stored purchased parts",
          len(section(cost_rows, "Part purchase prices")) - 1 == sum(1 for part in inputs["parts"] if part["sourcing"] == "purchased"),
          len(section(cost_rows, "Part purchase prices")) - 1)
    check(f"{kind}: the part attached cost lines follow the stored lines",
          len(section(cost_rows, "Part attached cost lines")) - 1 == sum(len(part["attached_costs"]) for part in inputs["parts"]),
          len(section(cost_rows, "Part attached cost lines")) - 1)
    check(f"{kind}: the product-root attached cost lines follow the stored lines",
          len(section(cost_rows, "Product root attached cost lines")) - 1 ==
          sum(len(product["root_attached_costs"]) for product in inputs["products"]),
          len(section(cost_rows, "Product root attached cost lines")) - 1)

    # -- Tooling ----------------------------------------------------------------------------
    tooling = workbook["Tooling"]
    tooling_rows = rows_of(tooling)
    tooling_totals = {row[0]: row[2] for row in tooling_rows if row and row[0] in
                      ("Project investment", "Project amortized recovered", "Project under/over recovery", "Project separate charge")}
    tooling_properties = {row[0]: row[1] for row in tooling_rows if row and row[0] in ("Treatment", "State")}
    if kind == "incomplete":
        check(f"{kind}: an unfinished project has no tooling treatment", tooling_properties.get("Treatment") == "not selected",
              tooling_properties)
    else:
        entries = tooling_input["tooling"]["entries"]
        expected_investment = sum(Decimal(str(entry["tool_count"])) * Decimal(entry["cost_per_tool"]) for entry in entries)
        expected_treatment = tooling_input["tooling"]["default_treatment"]
        check(f"{kind}: the tooling treatment is the stored project default",
              tooling_properties.get("Treatment") == expected_treatment, (tooling_properties.get("Treatment"), expected_treatment))
        check(f"{kind}: the tooling project investment reconciles to the stored tool entries",
              close(tooling_totals.get("Project investment"), expected_investment),
              (tooling_totals.get("Project investment"), expected_investment))
        check(f"{kind}: the tooling separate charge reconciles to the stored tool entries",
              close(tooling_totals.get("Project separate charge"), expected_investment),
              (tooling_totals.get("Project separate charge"), expected_investment))
        tools = section(tooling_rows, "Tools")
        check(f"{kind}: the tool rows follow the stored tool entries", len(tools) - 1 == len(entries), len(tools) - 1)
        allocations = section(tooling_rows, "Allocations")
        check(f"{kind}: the allocation rows follow the stored quoted products", len(allocations) - 1 == 2, len(allocations) - 1)

    # -- Scenarios --------------------------------------------------------------------------
    scenarios = workbook["Scenarios"]
    scenario_rows = rows_of(scenarios)
    summary_section = section(scenario_rows, "Scenario summary")
    scenario_data = summary_section[1:]
    check(f"{kind}: the scenario summary rows follow the requested saved scenarios",
          len(scenario_data) == len(scenario_inputs), (len(scenario_data), len(scenario_inputs)))
    detail_section = section(scenario_rows, "Scenario details")
    detail_data = detail_section[1:]
    years = sorted({forecast["year"] for product in inputs["products"] for forecast in product["forecasts"]})
    expected_detail_rows = len(scenario_inputs) * sum(len(product["forecasts"]) for product in inputs["products"]) if scenario_inputs else 0
    check(f"{kind}: the scenario detail rows cover every scenario x product x year",
          len(detail_data) == expected_detail_rows, (len(detail_data), expected_detail_rows))
    if scenario_inputs:
        by_id = {scenario["scenario_id"]: scenario for scenario in scenario_inputs}
        for row in scenario_data:
            stored = by_id.get(row[0])
            check(f"{kind}: every scenario summary row is a requested scenario", stored is not None, row[0])
            check(f"{kind}: the scenario name is the saved name", row[2] == stored["content"]["name"], (row[2], stored["content"]["name"]))
            check(f"{kind}: the scenario version is the saved version", row[1] == 1, row[1])
            check(f"{kind}: the scenario is directly comparable", row[3] == "no" and row[4] in (None, ""), row[3:5])
        price_by_product = {}
        for stored in scenario_inputs:
            for price in stored["content"]["manual_prices"]:
                price_by_product[(stored["scenario_id"], price["product_id"])] = Decimal(price["price"])
        for row in detail_data:
            scenario_name, version, product_id, product_name, revision_id, year = row[0], row[1], row[2], row[3], row[4], row[5]
            stored_scenario = next((stored for stored in scenario_inputs if stored["content"]["name"] == scenario_name), None)
            check(f"{kind}: every detail row names a requested scenario", stored_scenario is not None, scenario_name)
            base_price = price_by_product.get((stored_scenario["scenario_id"], product_id))
            check(f"{kind}: the detail row names a saved first-year manual price", base_price is not None,
                  (scenario_name, product_id))
            expected_manual = base_price * (Decimal(1) - ANNUAL_PRODUCTIVITY) ** (year - years[0])
            check(f"{kind}: the manual base price applies the stored annual productivity", close(row[9], expected_manual),
                  (scenario_name, product_id, year, row[9], expected_manual))
            check(f"{kind}: the manual total price is the manual base price for a separate tooling charge",
                  close(row[12], expected_manual), (scenario_name, product_id, year, row[12], expected_manual))
            forecast = dec(row[6])
            check(f"{kind}: the scenario revenue is forecast times the manual total price",
                  close(row[17], forecast * dec(row[12])), (scenario_name, product_id, year, row[17]))
            recurring = dec(products_out[product_id]["recurring_unit_cost"])
            check(f"{kind}: the scenario EBIT is forecast times the manual price margin",
                  close(row[18], forecast * (dec(row[12]) - recurring)), (scenario_name, product_id, year, row[18]))

    # -- Issues -----------------------------------------------------------------------------
    issues = workbook["Issues"]
    issue_rows = rows_of(issues)
    issue_data = [row for row in issue_rows[1:] if row[0] is not None]
    check(f"{kind}: the Summary issue count equals the Issues sheet rows", summary_map.get("Issue count") == len(issue_data),
          (summary_map.get("Issue count"), len(issue_data)))
    stored_issue_codes = {issue["code"] for issue in outputs["issues"]}
    workbook_issue_codes = {row[1] for row in issue_data}
    check(f"{kind}: every stored snapshot issue is printed", stored_issue_codes <= workbook_issue_codes,
          (sorted(stored_issue_codes - workbook_issue_codes)))
    if kind == "accepted":
        check(f"{kind}: the accepted history carries no listed issue", len(issue_data) == 0, issue_data)
    if kind == "draft":
        check(f"{kind}: the stale draft lists the staleness statement",
              ("staleness", "newer_shared_data") in {(row[0], row[1]) for row in issue_data}, issue_data)
        check(f"{kind}: the stale draft lists exactly one issue", len(issue_data) == 1, issue_data)
    if kind == "incomplete":
        expected_codes = stored_issue_codes | {"feasibility_not_assessed"}
        check(f"{kind}: the incomplete draft lists its missing inputs and findings", expected_codes <= workbook_issue_codes,
              (sorted(expected_codes - workbook_issue_codes)))
        check(f"{kind}: the incomplete draft lists four issues", len(issue_data) == 4, len(issue_data))

    # -- layout behaviour -------------------------------------------------------------------
    check(f"{kind}: the Summary keeps the key/value layout", summary.max_column == 2, summary.max_column)
    for sheet_name, frozen in [("Products", "A2"), ("BOM", "A2"), ("Routings", "A2"), ("Feasibility", "A2"), ("Issues", "A2")]:
        worksheet = workbook[sheet_name]
        check(f"{kind}: {sheet_name} keeps its header row", worksheet.max_row >= 1 and worksheet.cell(row=1, column=1).value is not None,
              worksheet.cell(row=1, column=1).value)
        if worksheet.max_row > 1:
            check(f"{kind}: {sheet_name} freezes the header row", worksheet.freeze_panes == frozen, worksheet.freeze_panes)
            check(f"{kind}: {sheet_name} carries an autofilter", worksheet.auto_filter.ref is not None, worksheet.auto_filter.ref)
            check(f"{kind}: {sheet_name} repeats the header row when printed",
                  getattr(worksheet, "print_title_rows", None) == "$1:$1", getattr(worksheet, "print_title_rows", None))
        else:
            # A table with no data row needs no frozen pane or filter; the header alone is the sheet.
            check(f"{kind}: the empty {sheet_name} table keeps exactly its header row", worksheet.max_row == 1, worksheet.max_row)
    for sheet_name in ("Summary", "Costs", "Tooling", "Scenarios"):
        check(f"{kind}: the {sheet_name} block sheet has no table autofilter", workbook[sheet_name].auto_filter.ref is None,
              workbook[sheet_name].auto_filter.ref)
    widths = workbook["Products"].column_dimensions
    check(f"{kind}: the long product-name column is wide", widths["B"].width == 30, widths["B"].width)
    check(f"{kind}: the long customer-part-number column is wide", widths["D"].width == 22, widths["D"].width)
    check(f"{kind}: the Issues message column is wide", workbook["Issues"].column_dimensions["D"].width == 100,
          workbook["Issues"].column_dimensions["D"].width)
    check(f"{kind}: the long product names wrap", workbook["Products"].cell(row=2, column=2).alignment.wrap_text is True,
          workbook["Products"].cell(row=2, column=2).alignment)
    if workbook["BOM"].max_row > 1:
        check(f"{kind}: the classification and part numbers wrap", workbook["BOM"].cell(row=2, column=4).alignment.wrap_text is True,
              workbook["BOM"].cell(row=2, column=4).alignment)
    if issue_data:
        check(f"{kind}: the issue messages wrap", workbook["Issues"].cell(row=2, column=4).alignment.wrap_text is True,
              workbook["Issues"].cell(row=2, column=4).alignment)
    expected_money = any(product_out[field] is not None for product_out in outputs["products"]
                         for field in ("material_cost", "production_cost", "attached_cost", "unit_total", "recurring_unit_cost"))
    expected_quantity = any(product_out["lifetime_demand"] is not None for product_out in outputs["products"]) or \
        any(product["forecasts"] for product in inputs["products"])
    if expected_money:
        check(f"{kind}: the Products money cells use the money number format", has_number_format(products, "#,##0.00") is not None,
              "no #,##0.00 cell")
    if expected_quantity:
        check(f"{kind}: the Products demand cells use the quantity number format", has_number_format(products, "#,##0") is not None,
              "no #,##0 cell")
    if outputs["annual"] and any(line["manufacturing_cost"] is not None for line in outputs["annual"]):
        check(f"{kind}: the Costs money cells use the money number format", has_number_format(costs, "#,##0.00") is not None,
              "no #,##0.00 cell")
    if kind != "incomplete":
        check(f"{kind}: the allocation share uses the percentage number format",
              has_number_format(workbook["Tooling"], "0.0000") is not None, "no 0.0000 cell")

    return {"worksheet_count": len(workbook.sheetnames), "products_rows": len(data), "bom_rows": len(bom_data),
            "routing_rows": len(routing_data), "feasibility_rows": len(feasibility_data), "cost_rows": len(cost_rows),
            "scenario_detail_rows": len(detail_data), "issue_rows": len(issue_data),
            "lifetime_manufacturing_total": str(lifetime) if lifetime is not None else None,
            "setup_total": str(setup_total) if setup_total is not None else None,
            "annual": [{"year": row[0], "manufacturing_cost": str(row[1])} for row in annual_data]}


def validate_pdf(kind, path, bundle, snapshot, evidence_dir, committed_base, scenario_inputs, tooling_input, render_dpi):
    metadata = bundle["metadata"]
    outputs = snapshot["outputs"]
    reader = PdfReader(path, strict=True)
    check(f"{kind}: the PDF strictly parses", reader is not None)
    check(f"{kind}: the PDF carries a PDF header", reader.pdf_header.startswith("%PDF-"), reader.pdf_header)
    pages = page_texts(reader)
    text = "\n".join(pages)
    check(f"{kind}: the PDF has pages", len(pages) >= 1, len(pages))
    check(f"{kind}: the PDF is the internal review document",
          "Internal export" in text and "Internal review material - not a customer quotation" in text, text[:200])
    check(f"{kind}: the PDF states the currency and its unit rule", "Currency: CNY - every amount below is stated in it" in text, text[:300])
    check(f"{kind}: the PDF states the WinAnsi limitation", "Characters outside the standard font's WinAnsi set are shown as '?'" in text)
    ascii_name = metadata["project_name"].encode("ascii", "ignore").decode()
    check(f"{kind}: the PDF names the project", "SB-21" in text and ascii_name.split("  ")[0][:30] in text, text[:300])
    check(f"{kind}: the PDF pins the calculation id", metadata["calculation"]["calculation_id"] in text, text[:600])
    check(f"{kind}: the PDF pins the project record version", f"v{metadata['versions']['project_version']} (pinned)" in text, text[:600])
    check(f"{kind}: the PDF pins the engineering record version", f"v{metadata['versions']['engineering_version']} (pinned)" in text, text[:600])
    check(f"{kind}: the PDF pins the manufacturing record version", f"v{metadata['versions']['manufacturing_version']} (pinned)" in text, text[:600])
    feasibility_state = "pinned" if metadata["versions"]["feasibility_pinned"] else "read at export time"
    check(f"{kind}: the PDF states how the feasibility record was read",
          f"v{metadata['versions']['feasibility_version']} ({feasibility_state})" in text, text[:800])
    check(f"{kind}: the PDF prints the calculator version", metadata["calculation"]["calculator_version"] in text)
    check(f"{kind}: the PDF prints the canonical input hash", metadata["calculation"]["input_hash"] in text)
    for label in metadata["labels"]:
        check(f"{kind}: the PDF prints the stable label {label}", label in text, text[:1200])
    for reason in metadata["stale_reasons"]:
        check(f"{kind}: the PDF prints the stale reason {reason}", reason.replace("_", " ") in text, text[:1200])
    if not metadata["labels"]:
        check(f"{kind}: a fresh export shows the fresh status band", "FRESH RESULT" in text, text[:400])
    for display, names in [
        ("DRAFT RESULT", {"draft_result"}), ("INCOMPLETE", {"incomplete_result"}), ("MISSING INPUTS", {"missing_inputs"}),
        ("UNRESOLVED FINDINGS", {"unresolved_findings"}), ("OUTDATED COSTS", {"outdated_costs"}),
        ("SUPERSEDED ACCEPTED BASELINE", {"superseded_accepted_baseline"}), ("STALE SCENARIOS", {"stale_scenarios"}),
        ("UNPINNED FEASIBILITY RECORD", {"unpinned_feasibility_record"}),
    ]:
        if names & set(metadata["labels"]):
            check(f"{kind}: the PDF status band shows {display}", display in text, text[:400])
    for section_name in ["Result metadata", "Cost summary", "Tooling and setup", "Unresolved issues and missing inputs"]:
        check(f"{kind}: the PDF has the {section_name} section", section_name in text, text[:1200])

    # Every page repeats the footer naming the selected result and the project; DRAFT/INCOMPLETE is
    # repeated in the footer as well.
    for index, page in enumerate(pages):
        check(f"{kind}: every PDF page carries the {index + 1} of {len(pages)} footer",
              f"Internal review only - page {index + 1} of {len(pages)}" in page, page[:300])
        check(f"{kind}: every PDF page carries the project footer", "Project: " in page, page[:300])
        if metadata["baseline"].get("draft_calculation"):
            check(f"{kind}: every PDF page carries the selected draft identity",
                  metadata["baseline"]["draft_calculation"]["calculation_id"] in page, page[-300:])
        if {"draft_result", "incomplete_result"} & set(metadata["labels"]):
            expected_footer = "DRAFT RESULT" in page or "INCOMPLETE" in page
            check(f"{kind}: the draft state is repeated in the page {index + 1} footer", expected_footer, page[-300:])

    # Money reconciliation against the stored snapshot, printed at two fractional digits.
    numbers = number_tokens(text)
    total = outputs["project_total_manufacturing_cost"]
    if total is None:
        check(f"{kind}: a missing lifetime total is printed as missing", "missing" in text.lower(), text[:600])
    else:
        check(f"{kind}: the PDF lifetime manufacturing total reconciles to the snapshot",
              shown_money(total) in numbers, (money_text(total), sorted(set(numbers))[:10]))
    setup_total = outputs["setup_total"]
    check(f"{kind}: the PDF separate setup total reconciles to the snapshot",
          shown_money(setup_total) in numbers, (money_text(setup_total), sorted(set(numbers))[:10]))
    for line in outputs["annual"]:
        if line["manufacturing_cost"] is None:
            check(f"{kind}: the missing annual cost for {line['year']} is marked missing", str(line["year"]) in text and "missing" in text.lower())
            continue
        shown = money_text(line["manufacturing_cost"])
        check(f"{kind}: the PDF annual cost for {line['year']} reconciles to the snapshot",
              Decimal(shown.replace(",", "")) in numbers, (line["year"], shown))

    # Tooling and setup sections.
    if kind == "incomplete":
        check(f"{kind}: the unfinished export states the missing tooling treatment", "not selected" in text, text[-1500:])
    else:
        entries = tooling_input["tooling"]["entries"]
        expected_investment = sum(Decimal(str(entry["tool_count"])) * Decimal(entry["cost_per_tool"]) for entry in entries)
        check(f"{kind}: the PDF tooling investment reconciles to the stored tool entries",
              shown_money(expected_investment) in numbers, (money_text(expected_investment), sorted(set(numbers))[:10]))
        check(f"{kind}: the PDF separate tooling charge reconciles to the stored tool entries",
              text.count(money_text(expected_investment)) >= 2, text.count(money_text(expected_investment)))
        check(f"{kind}: the PDF setup lines section lists the stored setup lines", "Separate setup lines" in text, text[-1200:])
        setup_line_count = outputs["setup"]
        check(f"{kind}: every stored setup line owner is printed",
              all(str(line["owner"].get("part_version_id") or line["owner"].get("product_id")) in text or
                  "Part " in text for line in setup_line_count), len(setup_line_count))

    # Commercial scenarios.
    if kind == "draft":
        check(f"{kind}: the PDF has one scenario block per saved scenario",
              text.count("Scenario: ") == len(scenario_inputs), (text.count("Scenario: "), len(scenario_inputs)))
        for stored in scenario_inputs:
            check(f"{kind}: the PDF prints the saved scenario name", stored["content"]["name"] in text, stored["content"]["name"])
        check(f"{kind}: the PDF prints the lifetime revenue label", "Lifetime revenue" in text)
        check(f"{kind}: the PDF prints the lifetime EBIT label", "Lifetime EBIT" in text)
        detail_headers = ["Product", "Year", "Forecast", "Manual price", "Revenue", "EBIT amount", "Suggested price"]
        detail_pages = [index for index, page in enumerate(pages) if "Manual price" in page]
        check(f"{kind}: the scenario detail table is paginated across pages", len(detail_pages) >= 2, detail_pages)
        for index in detail_pages:
            for header in detail_headers:
                check(f"{kind}: the scenario table header {header!r} repeats on page {index + 1}", header in pages[index],
                      pages[index][:400])
        check(f"{kind}: every saved first-year manual price is printed",
              all(Decimal(price["price"]).quantize(Decimal("0.01")) in numbers
                  for stored in scenario_inputs for price in stored["content"]["manual_prices"]),
              "a saved manual price is missing")
    else:
        check(f"{kind}: no scenario section is rendered when none is requested",
              "No saved commercial scenario is part of this export." in text, text[-1500:])

    # The issue-count line uses the singular for exactly one listed entry and the plural otherwise
    # (review-fix revision: the first capture printed "1 listed entries.").
    if kind == "draft":
        check(f"{kind}: the single listed issue uses singular grammar", "1 listed entry." in text, text[-1500:])
        check(f"{kind}: the single listed issue does not use the plural grammar", "1 listed entries." not in text, text[-1500:])
    if kind == "incomplete":
        check(f"{kind}: the four listed issues use plural grammar", "4 listed entries." in text, text[-1500:])
    if kind == "accepted":
        check(f"{kind}: the accepted history states that no issue is recorded",
              "No unresolved issue or missing input is recorded." in text, text[-1500:])

    # Broken text: no U+FFFD, no raw control characters, and the non-WinAnsi product name is replaced
    # by '?' rather than raw UTF-8 (the workbook carries the exact text).
    check(f"{kind}: the PDF text has no replacement character", "\ufffd" not in text)
    controls = [character for character in text if ord(character) < 32 and character not in "\n\r\t"]
    check(f"{kind}: the PDF text has no raw control character", not controls, controls[:8])
    check(f"{kind}: the non-WinAnsi product name is not raw UTF-8 in the PDF", "支架" not in text)
    check(f"{kind}: the non-WinAnsi product name is replaced by question marks", "?" in text)

    # Render every page for the visual inspection.
    rendered = []
    document = pymupdf.open(path)
    try:
        for index, page in enumerate(document):
            pixmap = page.get_pixmap(dpi=render_dpi)
            target = os.path.join(evidence_dir, f"{committed_base}-page-{index + 1}.png")
            pixmap.save(target)
            rendered.append({"page": index + 1, "file": f"{committed_base}-page-{index + 1}.png",
                             "bytes": os.path.getsize(target), "width": pixmap.width, "height": pixmap.height})
    finally:
        document.close()
    for page in rendered:
        check(f"{kind}: the rendered page {page['page']} PNG exists and is readable",
              os.path.getsize(os.path.join(evidence_dir, page["file"])) > 5000, page)

    return {"pages": len(pages), "rendered": rendered, "money_facts": str(len(set(numbers))),
            "scenario_blocks": text.count("Scenario: ")}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", default="/tmp/sb21_evidence")
    parser.add_argument("--evidence", default=os.path.join(os.getcwd(), "docs/solidbom/execution/evidence"))
    parser.add_argument("--dpi", type=int, default=130)
    args = parser.parse_args()
    os.makedirs(args.evidence, exist_ok=True)

    captures = load_json(os.path.join(args.out, "captures.json"))
    summary = captures["summary"]
    documents = captures["documents"]
    capture_by_name = {capture["capture"]: capture for capture in captures["captures"]}

    scenario_inputs = [{"scenario_id": capture_by_name[f"scenario-saved-{index}"]["request"]["scenario_id"],
                        "content": capture_by_name[f"scenario-saved-{index}"]["request"]["content"]}
                       for index in range(3)]
    tooling_input = capture_by_name["tooling-settings"]["request"]

    snapshots = {
        "draft": load_json(os.path.join(args.out, "snapshot-draft-1.json")),
        "accepted": load_json(os.path.join(args.out, "snapshot-accepted-1.json")),
        "incomplete": load_json(os.path.join(args.out, "snapshot-incomplete.json")),
    }

    examples = [
        {"kind": "draft", "xlsx": documents["draft-outdated-xlsx"], "pdf": documents["draft-outdated-pdf"],
         "committed": "SB-21-draft-export", "snapshot": "draft"},
        {"kind": "accepted", "xlsx": documents["accepted-superseded-xlsx"], "pdf": documents["accepted-superseded-pdf"],
         "committed": "SB-21-accepted-export", "snapshot": "accepted"},
        {"kind": "incomplete", "xlsx": documents["incomplete-draft-xlsx"], "pdf": documents["incomplete-draft-pdf"],
         "committed": "SB-21-incomplete-draft-export", "snapshot": "incomplete"},
    ]

    import openpyxl as openpyxl_module
    report = {
        "environment": {
            "host_platform": platform.platform(),
            "python": sys.version.split()[0],
            "openpyxl": openpyxl_module.__version__,
            "pypdf": __import__("pypdf").__version__,
            "pymupdf": pymupdf.__version__,
            "render_dpi": args.dpi,
        },
        "examples": [], "comparisons": [], "rendered_files": [], "checks": CHECKS,
    }
    workbooks = {}
    for example in examples:
        kind = example["kind"]
        snapshot = snapshots[example["snapshot"]]
        xlsx_path = os.path.join(args.out, f"{example['xlsx']['label']}.xlsx")
        pdf_path = os.path.join(args.out, f"{example['pdf']['label']}.pdf")
        example_scenarios = scenario_inputs if kind == "draft" else []
        workbook_report = validate_workbook(kind, xlsx_path, example["xlsx"], snapshot, captures, example_scenarios, tooling_input)
        pdf_report = validate_pdf(kind, pdf_path, example["pdf"], snapshot, args.evidence, example["committed"],
                                 example_scenarios, tooling_input, args.dpi)
        workbooks[kind] = {"path": xlsx_path, "book": openpyxl.load_workbook(xlsx_path)}
        layout_book = workbooks[kind]["book"]
        layout = {
            "dimensions": {worksheet.title: worksheet.dimensions for worksheet in layout_book.worksheets},
            "freeze_panes": {worksheet.title: worksheet.freeze_panes for worksheet in layout_book.worksheets},
            "autofilter": {worksheet.title: worksheet.auto_filter.ref for worksheet in layout_book.worksheets},
            "print_titles": {worksheet.title: getattr(worksheet, "print_title_rows", None) for worksheet in layout_book.worksheets},
            "row_heights": {
                "Products": [layout_book["Products"].row_dimensions[index].height
                             for index in range(1, layout_book["Products"].max_row + 1)],
                "BOM": [layout_book["BOM"].row_dimensions[index].height
                        for index in range(1, layout_book["BOM"].max_row + 1)],
                "Issues": [layout_book["Issues"].row_dimensions[index].height
                           for index in range(1, layout_book["Issues"].max_row + 1)],
            },
            "column_widths": {
                sheet_name: [layout_book[sheet_name].column_dimensions[column].width
                             for column in ("A", "B", "C", "D")]
                for sheet_name in ("Summary", "Products", "BOM", "Issues")
            },
            "number_formats": sorted({cell.number_format for worksheet in layout_book.worksheets
                                      for row in worksheet.iter_rows() for cell in row
                                      if cell.value is not None}),
            "wrapped_cells": sum(1 for worksheet in layout_book.worksheets
                                 for row in worksheet.iter_rows() for cell in row
                                 if cell.alignment is not None and cell.alignment.wrap_text is True),
        }
        report["examples"].append({
            "kind": kind,
            "committed_base": example["committed"],
            "file_name": example["pdf"]["file_name"],
            "labels": example["pdf"]["metadata"]["labels"],
            "xlsx_bytes": os.path.getsize(xlsx_path),
            "pdf_bytes": os.path.getsize(pdf_path),
            "pdf_sha256": example["pdf"]["sha256"],
            "xlsx_sha256": example["xlsx"]["sha256"],
            "workbook": workbook_report,
            "pdf": pdf_report,
        })
        report["examples"][-1]["layout"] = layout
        report["rendered_files"].extend(pdf_report["rendered"])

    # The clean and the after-change exports of the same stored result carry the same cost facts.
    for kind, clean_xlsx, changed_xlsx, clean_pdf, changed_pdf in [
        ("draft", "draft-clean-xlsx", "draft-outdated-xlsx", "draft-clean-pdf", "draft-outdated-pdf"),
        ("accepted", "accepted-before-second-activation-xlsx", "accepted-superseded-xlsx",
         "accepted-before-second-activation-pdf", "accepted-superseded-pdf"),
    ]:
        clean_doc = documents[clean_xlsx]
        changed_doc = documents[changed_xlsx]
        check(f"{kind}: the clean and after-change exports name the same calculation id",
              clean_doc["metadata"]["calculation"]["calculation_id"] == changed_doc["metadata"]["calculation"]["calculation_id"],
              (clean_doc["metadata"]["calculation"]["calculation_id"], changed_doc["metadata"]["calculation"]["calculation_id"]))
        check(f"{kind}: the clean and after-change exports pin the same record versions",
              clean_doc["metadata"]["versions"] == changed_doc["metadata"]["versions"],
              (clean_doc["metadata"]["versions"], changed_doc["metadata"]["versions"]))
        clean_book = openpyxl.load_workbook(os.path.join(args.out, f"{clean_xlsx}.xlsx"))
        changed_book = workbooks[kind]["book"]
        check_pairs(f"{kind}: the clean and after-change workbooks carry the same numeric facts",
                    workbook_values(clean_book), workbook_values(changed_book))
        _, clean_text = None, "\n".join(page_texts(PdfReader(os.path.join(args.out, f"{clean_pdf}.pdf"), strict=True)))
        _, changed_text = None, "\n".join(page_texts(PdfReader(os.path.join(args.out, f"{changed_pdf}.pdf"), strict=True)))
        check_pairs(f"{kind}: the clean and after-change PDFs carry the same money facts",
                    [str(token) for token in number_tokens(clean_text)], [str(token) for token in number_tokens(changed_text)])
        report["comparisons"].append({"kind": kind, "xlsx_numeric_cells": len(workbook_values(clean_book)),
                                      "pdf_money_facts": len(set(number_tokens(clean_text)))})

    # The accepted snapshot still names the shared-data versions that were current when it was read,
    # not the newer ones published later: accepted history was not re-resolved from current master data.
    accepted_snapshot = snapshots["accepted"]
    complete = summary["complete"]
    export_summary = summary["export"]
    material_ids = {part["material"]["material_version_id"] for part in accepted_snapshot["inputs"]["parts"] if part["material"]}
    check("the accepted snapshot still names the older material version",
          material_ids == {complete["material_v1"]}, (material_ids, complete["material_v1"]))
    check("a newer material version was published after the accepted read",
          export_summary["material_v2"] != complete["material_v1"], (export_summary["material_v2"], complete["material_v1"]))
    rate_ids = {part["operations"][0]["cycle"]["production_rate_version_id"]
                for part in accepted_snapshot["inputs"]["parts"] if part["operations"]}
    check("the accepted snapshot still names the older production rate version",
          rate_ids == {complete["production_rate_v1"]}, (rate_ids, complete["production_rate_v1"]))
    check("a newer production rate version was published after the accepted read",
          export_summary["rate_v2"] != complete["production_rate_v1"],
          (export_summary["rate_v2"], complete["production_rate_v1"]))

    print(json.dumps(report, indent=2, default=str))
    with open(os.path.join(args.out, "validation-report.json"), "w", encoding="utf-8") as handle:
        json.dump(report, handle, indent=2, default=str)
        handle.write("\n")
    print(f"\n{len(CHECKS)} independent validation checks passed")


if __name__ == "__main__":
    main()
